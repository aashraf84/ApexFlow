   148→function StatusBadge({ status }: { status: string }) {
   149→  const config: Record<string, { label: string; className: string; style?: React.CSSProperties }> = {
   150→    active: {
   151→      label: 'Active',
   152→      className: '',
   153→      style: {
   154→        background: 'color-mix(in srgb, var(--gg-vermillion) 15%, transparent)',
   155→        color: 'var(--gg-vermillion)',
   156→        borderColor: 'color-mix(in srgb, var(--gg-vermillion) 20%, transparent)',
   157→      },
   158→    },
   159→    inactive: { label: 'Inactive', className: 'bg-red-500/15 text-red-400 border-red-500/20' },
   160→    prospect: { label: 'Prospect', className: 'bg-amber-500/15 text-amber-400 border-amber-500/20' },
   161→    churned: { label: 'Churned', className: 'bg-gray-500/15 text-gray-400 border-gray-500/20' },
   162→  }
   163→  const c = config[status] || config.prospect
   164→  return (
   165→    <Badge variant="outline" className={`text-xs ${c.className}`} style={c.style}>
   166→      {c.label}
   167→    </Badge>
   168→  )
   169→}

   248→// ============================================
   249→// Main Component
   250→// ============================================
   251→export default function AdminClientsPage() {
   252→  // State
   253→  const [clients, setClients] = useState<ClientData[]>([])
   254→  const [loading, setLoading] = useState(true)
   255→  const [search, setSearch] = useState('')
   256→  const [statusFilter, setStatusFilter] = useState('all')
   257→
   258→  // View state
   259→  const [selectedClient, setSelectedClient] = useState<ClientData | null>(null)
   260→  const [detailLoading, setDetailLoading] = useState(false)
   261→
   262→  // Dialog state
   263→  const [clientDialogOpen, setClientDialogOpen] = useState(false)
   264→  const [editingClient, setEditingClient] = useState<ClientData | null>(null)
   265→  const [clientForm, setClientForm] = useState<ClientFormData>(emptyClientForm)
   266→  const [submitting, setSubmitting] = useState(false)
   267→
   268→  // Delete dialog
   269→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   270→  const [deletingClient, setDeletingClient] = useState<ClientData | null>(null)
   271→
   272→  // Contact dialog
   273→  const [contactDialogOpen, setContactDialogOpen] = useState(false)
   274→  const [contactForm, setContactForm] = useState({
   275→    name: '',
   276→    email: '',
   277→    phone: '',
   278→    role: '',
   279→    isPrimary: false,
   280→    notes: '',
   281→  })
   282→
   283→  // Activity dialog
   284→  const [activityDialogOpen, setActivityDialogOpen] = useState(false)
   285→  const [activityForm, setActivityForm] = useState({
   286→    type: 'note',
   287→    subject: '',
   288→    notes: '',
   289→  })
   290→
   291→  // Contact delete
   292→  const [deleteContactDialogOpen, setDeleteContactDialogOpen] = useState(false)
   293→  const [deletingContactId, setDeletingContactId] = useState<string | null>(null)
   294→
   295→  // ============================================
   296→  // Fetch clients
   297→  // ============================================
   298→  const fetchClients = useCallback(async () => {
   299→    try {
   300→      setLoading(true)
   301→      const params = new URLSearchParams()
   302→      if (statusFilter && statusFilter !== 'all') params.set('status', statusFilter)
   303→      if (search) params.set('search', search)
   304→      params.set('pageSize', '100')
   305→
   306→      const res = await fetch(`/api/clients?${params.toString()}`)
   307→      if (res.ok) {
   308→        const json = await res.json()
   309→        setClients(json.data || [])
   310→      }
   311→    } catch {
   312→      toast.error('Failed to load clients')
   313→    } finally {
   314→      setLoading(false)
   315→    }
   316→  }, [search, statusFilter])
   317→
   318→  useEffect(() => {
   319→    fetchClients()
   320→  }, [fetchClients])
   321→
   322→  // ============================================
   323→  // Fetch client detail
   324→  // ============================================
   325→  const fetchClientDetail = async (id: string) => {
   326→    try {
   327→      setDetailLoading(true)
   328→      const res = await fetch(`/api/clients/${id}`)
   329→      if (res.ok) {
   330→        const json = await res.json()
   331→        setSelectedClient(json.data)
   332→      } else {
   333→        toast.error('Failed to load client details')
   334→      }
   335→    } catch {
   336→      toast.error('Failed to load client details')
   337→    } finally {
   338→      setDetailLoading(false)
   339→    }
   340→  }
   341→
   342→  // ============================================
   343→  // Client CRUD
   344→  // ============================================
   345→  const openCreateDialog = () => {
   346→    setEditingClient(null)
   347→    setClientForm(emptyClientForm)
   348→    setClientDialogOpen(true)
   349→  }
   350→
   351→  const openEditDialog = (client: ClientData) => {
   352→    setEditingClient(client)
   353→    setClientForm({
   354→      name: client.name,
   355→      email: client.email || '',
   356→      phone: client.phone || '',
   357→      company: client.company || '',
   358→      address: client.address || '',
   359→      city: client.city || '',
   360→      state: client.state || '',
   361→      country: client.country || '',
   362→      zipCode: client.zipCode || '',
   363→      website: client.website || '',
   364→      industry: client.industry || '',
   365→      status: client.status,
   366→      notes: client.notes || '',
   367→    })
   368→    setClientDialogOpen(true)
   369→  }
   370→
   371→  const handleClientSubmit = async () => {
   372→    if (!clientForm.name.trim()) {
   373→      toast.error('Client name is required')
   374→      return
   375→    }
   376→
   377→    try {
   378→      setSubmitting(true)
   379→
   380→      if (editingClient) {
   381→        // Update
   382→        const res = await fetch(`/api/clients/${editingClient.id}`, {
   383→          method: 'PUT',
   384→          headers: { 'Content-Type': 'application/json' },
   385→          body: JSON.stringify(clientForm),
   386→        })
   387→        if (res.ok) {
   388→          toast.success('Client updated successfully')
   389→          setClientDialogOpen(false)
   390→          fetchClients()
   391→          if (selectedClient?.id === editingClient.id) {
   392→            fetchClientDetail(editingClient.id)
   393→          }
   394→        } else {
   395→          const json = await res.json()
   396→          toast.error(json.error || 'Failed to update client')
   397→        }
   398→      } else {
   399→        // Create
   400→        const res = await fetch('/api/clients', {
   401→          method: 'POST',
   402→          headers: { 'Content-Type': 'application/json' },
   403→          body: JSON.stringify(clientForm),
   404→        })
   405→        if (res.ok) {
   406→          toast.success('Client created successfully')
   407→          setClientDialogOpen(false)
   408→          fetchClients()
   409→        } else {
   410→          const json = await res.json()
   411→          toast.error(json.error || 'Failed to create client')
   412→        }
   413→      }
   414→    } catch {
   415→      toast.error('An error occurred')
   416→    } finally {
   417→      setSubmitting(false)
   418→    }
   419→  }
   420→
   421→  const handleDeleteClient = async () => {
   422→    if (!deletingClient) return
   423→    try {
   424→      const res = await fetch(`/api/clients/${deletingClient.id}`, { method: 'DELETE' })
   425→      if (res.ok) {
   426→        toast.success('Client deleted successfully')
   427→        setDeleteDialogOpen(false)
   428→        setDeletingClient(null)
   429→        if (selectedClient?.id === deletingClient.id) {
   430→          setSelectedClient(null)
   431→        }
   432→        fetchClients()
   433→      } else {
   434→        toast.error('Failed to delete client')
   435→      }
   436→    } catch {
   437→      toast.error('An error occurred')
   438→    }
   439→  }
   440→
   441→  // ============================================
   442→  // Contact CRUD
   443→  // ============================================
   444→  const handleAddContact = async () => {
   445→    if (!selectedClient || !contactForm.name.trim()) {
   446→      toast.error('Contact name is required')
   447→      return
   448→    }
   449→
   450→    try {
   451→      setSubmitting(true)
   452→      const res = await fetch(`/api/clients/${selectedClient.id}/contacts`, {
   453→        method: 'POST',
   454→        headers: { 'Content-Type': 'application/json' },
   455→        body: JSON.stringify(contactForm),
   456→      })
   457→      if (res.ok) {
   458→        toast.success('Contact added successfully')
   459→        setContactDialogOpen(false)
   460→        setContactForm({ name: '', email: '', phone: '', role: '', isPrimary: false, notes: '' })
   461→        fetchClientDetail(selectedClient.id)
   462→      } else {
   463→        toast.error('Failed to add contact')
   464→      }
   465→    } catch {
   466→      toast.error('An error occurred')
   467→    } finally {
   468→      setSubmitting(false)
   469→    }
   470→  }
   471→
   472→  const handleDeleteContact = async () => {
   473→    if (!selectedClient || !deletingContactId) return
   474→    try {
   475→      const res = await fetch(
   476→        `/api/clients/${selectedClient.id}/contacts?contactId=${deletingContactId}`,
   477→        { method: 'DELETE' }
   478→      )
   479→      if (res.ok) {
   480→        toast.success('Contact deleted')
   481→        setDeleteContactDialogOpen(false)
   482→        setDeletingContactId(null)
   483→        fetchClientDetail(selectedClient.id)
   484→      } else {
   485→        toast.error('Failed to delete contact')
   486→      }
   487→    } catch {
   488→      toast.error('An error occurred')
   489→    }
   490→  }
   491→
   492→  // ============================================
   493→  // Activity CRUD
   494→  // ============================================
   495→  const handleAddActivity = async () => {
   496→    if (!selectedClient || !activityForm.subject.trim()) {
   497→      toast.error('Subject is required')
   498→      return
   499→    }
   500→
   501→    try {
   502→      setSubmitting(true)
   503→      const res = await fetch(`/api/clients/${selectedClient.id}/activities`, {
   504→        method: 'POST',
   505→        headers: { 'Content-Type': 'application/json' },
   506→        body: JSON.stringify(activityForm),
   507→      })
   508→      if (res.ok) {
   509→        toast.success('Activity added successfully')
   510→        setActivityDialogOpen(false)
   511→        setActivityForm({ type: 'note', subject: '', notes: '' })
   512→        fetchClientDetail(selectedClient.id)
   513→      } else {
   514→        toast.error('Failed to add activity')
   515→      }
   516→    } catch {
   517→      toast.error('An error occurred')
   518→    } finally {
   519→      setSubmitting(false)
   520→    }
   521→  }
   522→
   523→  // ============================================
   524→  // Format date
   525→  // ============================================
   526→  const formatDate = (dateStr: string) => {
   527→    try {
   528→      return new Date(dateStr).toLocaleDateString('en-US', {
   529→        year: 'numeric',
   530→        month: 'short',
   531→        day: 'numeric',
   532→      })
   533→    } catch {
   534→      return dateStr
   535→    }
   536→  }
   537→
   538→  const formatDateTime = (dateStr: string) => {
   539→    try {
   540→      return new Date(dateStr).toLocaleDateString('en-US', {
   541→        year: 'numeric',
   542→        month: 'short',
   543→        day: 'numeric',
   544→        hour: '2-digit',
   545→        minute: '2-digit',
   546→      })
   547→    } catch {
   548→      return dateStr
   549→    }
   550→  }
   551→
   552→  // ============================================
   553→  // Render: Loading skeletons
   554→  // ============================================
   555→  const renderSkeletons = () => (
   556→    <div className="space-y-3">
   557→      {Array.from({ length: 5 }).map((_, i) => (
   558→        <div key={i} className="flex items-center gap-4 p-4 rounded-lg bg-white/[0.02]">
   559→          <Skeleton className="h-10 w-10 rounded-full bg-white/5" />
   560→          <div className="flex-1 space-y-2">
   561→            <Skeleton className="h-4 w-40 bg-white/5" />
   562→            <Skeleton className="h-3 w-56 bg-white/5" />
   563→          </div>
   564→          <Skeleton className="h-6 w-20 bg-white/5" />
   565→        </div>
   566→      ))}
   567→    </div>
   568→  )
   569→
   570→  // ============================================
   571→  // RENDER: Client List View
   572→  // ============================================
   573→  const renderClientList = () => (
   574→    <div className="space-y-6">
   575→      {/* Header */}
   576→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   577→        <div>
   578→          <h2 className="text-2xl font-bold text-foreground">Clients</h2>
   579→          <p className="text-muted-foreground mt-1">Manage your client relationships and CRM</p>
   580→        </div>
   581→        <Button
   582→          onClick={openCreateDialog}
   583→          className="hover:opacity-90 text-white"
   584→          style={{ background: 'linear-gradient(135deg, var(--gg-vermillion), var(--gg-amber))' }}
   585→        >
   586→          <Plus className="h-4 w-4 mr-2" />
   587→          Add Client
   588→        </Button>
   589→      </div>
   590→
   591→      {/* Search & Filter */}
   592→      <div className="flex flex-col sm:flex-row gap-3">
   593→        <div className="relative flex-1">
   594→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-gray-500" />
   595→          <Input
   596→            placeholder="Search clients by name, email, company..."
   597→            value={search}
   598→            onChange={(e) => setSearch(e.target.value)}
   599→            className="pl-10 text-foreground placeholder:text-gray-500"
   600→            style={{ background: 'var(--card)', borderColor: 'var(--gg-separator)' }}
   601→          />
   602→        </div>
   603→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   604→          <SelectTrigger
   605→            className="w-full sm:w-44 text-foreground"
   606→            style={{ background: 'var(--card)', borderColor: 'var(--gg-separator)' }}
   607→          >
   608→            <SelectValue placeholder="Filter status" />
   609→          </SelectTrigger>
   610→          <SelectContent>
   611→            <SelectItem value="all">All Statuses</SelectItem>
   612→            <SelectItem value="active">Active</SelectItem>
   613→            <SelectItem value="inactive">Inactive</SelectItem>
   614→            <SelectItem value="prospect">Prospect</SelectItem>
   615→            <SelectItem value="churned">Churned</SelectItem>
   616→          </SelectContent>
   617→        </Select>
   618→      </div>
   619→
   620→      {/* Client Table / Cards */}
   621→      <Card
   622→        style={{ background: 'var(--card)', borderColor: 'var(--gg-separator)', boxShadow: 'var(--gg-shadow-card)' }}
   623→      >
   624→        <CardContent className="p-0">
   625→          {loading ? (
   626→            <div className="p-4">{renderSkeletons()}</div>
   627→          ) : clients.length === 0 ? (
   628→            <div className="flex flex-col items-center justify-center py-16">
   629→              <Users className="h-12 w-12 text-gray-600 mb-4" />
   630→              <p className="text-muted-foreground text-lg font-medium">No clients found</p>
   631→              <p className="text-gray-500 text-sm mt-1">
   632→                {search || statusFilter !== 'all'
   633→                  ? 'Try adjusting your filters'
   634→                  : 'Get started by adding your first client'}
   635→              </p>
   636→              {!search && statusFilter === 'all' && (
   637→                <Button
