   384→/* ────────────────────────────────────────────────────────── */
   385→/*  React Hook: Fetch ERP systems from API                   */
   386→/* ────────────────────────────────────────────────────────── */
   387→
   388→export function useErpSystems(status?: string) {
   389→  const [erps, setErps] = useState<ErpSystemData[]>([])
   390→  const [loading, setLoading] = useState(true)
   391→  const [error, setError] = useState<string | null>(null)
   392→
   393→  useEffect(() => {
   394→    const fetchErps = async () => {
   395→      try {
   396→        setLoading(true)
   397→        const url = status ? `/api/erp-systems?status=${status}` : '/api/erp-systems'
   398→        const res = await fetch(url)
   399→        const json = await res.json()
   400→        if (json.success && Array.isArray(json.data) && json.data.length > 0) {
   401→          setErps(json.data)
   402→        } else {
   403→          // API returned empty or failed — use fallback
   404→          const fallback = status
   405→            ? FALLBACK_ERP_DATA.filter((e) => e.status === status)
   406→            : FALLBACK_ERP_DATA
   407→          setErps(fallback)
   408→          if (!json.success) {
   409→            setError(json.message || 'Failed to fetch ERP systems')
   410→          }
   411→        }
   412→      } catch (err) {
   413→        // Network error — use hardcoded fallback so activeErps.length is never 0
   414→        const fallback = status
   415→          ? FALLBACK_ERP_DATA.filter((e) => e.status === status)
   416→          : FALLBACK_ERP_DATA
   417→        setErps(fallback)
   418→        setError(err instanceof Error ? err.message : 'Network error')
   419→      } finally {
   420→        setLoading(false)
   421→      }
   422→    }
   423→    fetchErps()
   424→  }, [status])
   425→
   426→  const activeErps = erps.filter((e) => e.status === 'active')
   427→  const comingSoonErps = erps.filter((e) => e.status === 'coming_soon')
   428→
   429→  // Dynamic industry verticals count based on unique industries from active ERPs
   430→  const industryVerticals = new Set(activeErps.map((e) => e.industry)).size
   431→
   432→  return { erps, activeErps, comingSoonErps, industryVerticals, loading, error }
   433→}
