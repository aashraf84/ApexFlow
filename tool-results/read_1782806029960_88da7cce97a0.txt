    39→// ─── Database URL Cache (from ErpSystem table) ──────────────────────────────
    40→let dbUrlCache: Record<string, string> | null = null
    41→let dbUrlCacheTs = 0
    42→const DB_URL_CACHE_TTL = 30_000 // 30 seconds
    43→
    44→async function loadErpDbUrls(): Promise<Record<string, string>> {
    45→  const now = Date.now()
    46→  if (dbUrlCache && now - dbUrlCacheTs < DB_URL_CACHE_TTL) return dbUrlCache
    47→
    48→  try {
    49→    const erps = await db.erpSystem.findMany({
    50→      select: { slug: true, databaseUrl: true },
    51→    })
    52→    const urls: Record<string, string> = {}
    53→    for (const erp of erps) {
    54→      if (erp.databaseUrl) urls[erp.slug] = erp.databaseUrl
    55→    }
    56→    dbUrlCache = urls
    57→    dbUrlCacheTs = now
    58→    return urls
    59→  } catch (err) {
    60→    console.warn('[ERP-CONNECTOR] Failed to load DB URLs from database:', err)
    61→    return dbUrlCache || {}
    62→  }
    63→}
    64→
    65→/** Get the database URL for an ERP slug — checks database first, falls back to env var */
    66→async function getErpDbUrl(slug: string): Promise<string> {
    67→  const dbUrls = await loadErpDbUrls()
    68→  return dbUrls[slug] || ERP_DB_URLS[slug] || ''
    69→}

   151→    const sql = neon(url, { fetchConnectionCache: true })
   152→    const [result] = await withTimeout(sql`SELECT 1 as ok`, 5_000)
   153→    const latency = Date.now() - start

   155→    if (result?.ok === 1) {

   169→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "tenants"`

   172→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "Client"`
   173→          clientCount = cnt?.c ?? 0

   175→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "ERPTenant"`
   176→          clientCount = cnt?.c ?? 0
   177→        } else if (slug === 'travelflow') {
   178→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "OwnerClient"`

   181→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "Client"`
   182→          clientCount = cnt?.c ?? 0
   183→        } else if (slug === 'factoryflow') {
   184→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "OwnerClient"`
   185→          clientCount = cnt?.c ?? 0
   186→        } else if (slug === 'foodflow') {
   187→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "Tenant"`
   188→          clientCount = cnt?.c ?? 0
   189→        } else if (slug === 'autoflow') {
   190→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "Client"`
   191→          clientCount = cnt?.c ?? 0
   192→        } else if (slug === 'healthflow') {
   193→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "Client"`
   194→          clientCount = cnt?.c ?? 0
   195→        } else if (slug === 'textileflow') {
   196→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "owner_clients"`
   197→          clientCount = cnt?.c ?? 0
   198→        } else if (slug === 'creativeflow') {
   199→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "erp_clients"`
   200→          clientCount = cnt?.c ?? 0
   201→        } else if (slug === 'distriflow') {
   202→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "OwnerClient"`
   203→          clientCount = cnt?.c ?? 0
   204→        } else if (slug === 'logisticflow') {
   205→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "Tenant"`
   206→          clientCount = cnt?.c ?? 0
   207→        } else if (slug === 'fleetflow') {
   208→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "Tenant"`

   211→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "Tenant"`

   214→          const [cnt] = await sql`SELECT COUNT(*)::int AS c FROM "OwnerClient"`

   217→      } catch (err) {
   218→        const msg = err instanceof Error ? err.message : String(err)
   219→        console.warn(`[ERP-CONNECTOR] ${slug} schema warning: ${msg}`)
   220→        // Connection IS working — just the client count table is different
   221→        // Return connected: true with a warning instead of connected: false
   222→      }

   224→        ? { connected: true, latency, clientCount }
   225→        : { connected: true, latency, clientCount: 0, schemaWarning: `Schema mismatch: could not count clients — table may differ from expected schema` }
   226→    }
   227→    return { connected: false, error: 'Unexpected response' }
   228→  } catch (err) {
   229→    return {
   230→      connected: false,
   231→      error: err instanceof Error ? err.message : 'Connection failed',
   232→    }
   233→  }
   234→}

   236→// ─── Timeout helper ──────────────────────────────────────────────────────────
   237→function withTimeout<T>(promise: Promise<T>, ms: number): Promise<T> {
   238→  return Promise.race([
   239→    promise,
   240→    new Promise<never>((_, reject) =>
   241→      setTimeout(() => reject(new Error(`Connection timeout after ${ms}ms`)), ms)
   242→    ),
   243→  ])
   244→}

   246→// ─── Get SQL for an ERP ──────────────────────────────────────────────────────
   247→async function getErpSql(slug: string): Promise<NeonQueryFunction<false, false> | null> {
   248→  const url = await getErpDbUrl(slug)
   249→  if (!url) return null
   250→  return neon(url, { fetchConnectionCache: true })
   251→}
   252→
   253→// ─── Connection cache (avoids re-testing recently-failed connections) ──────────
   254→const connectionCache = new Map<string, { ok: boolean; ts: number }>()
   255→const CACHE_TTL_MS = 60_000 // 1 minute for successful connections
   256→const FAILED_CACHE_TTL_MS = 30_000 // 30 seconds for failed connections (retry sooner)
   257→
   258→/** Quick connectivity check — returns true if the ERP database is reachable.
   259→ *  Caches successful results for 1 minute, failed results for 30 seconds
   260→ *  so we retry unreachable databases sooner. Uses a 5-second timeout to avoid
   261→ *  long waits on unresponsive hosts. */
   262→async function isErpReachable(slug: string): Promise<boolean> {
   263→  const cached = connectionCache.get(slug)
   264→  const ttl = cached?.ok ? CACHE_TTL_MS : FAILED_CACHE_TTL_MS
   265→  if (cached && Date.now() - cached.ts < ttl) return cached.ok
   266→
   267→  const url = await getErpDbUrl(slug)
   268→  if (!url) {
   269→    connectionCache.set(slug, { ok: false, ts: Date.now() })
   270→    return false
   271→  }
   272→
   273→  try {
   274→    const sql = neon(url, { fetchConnectionCache: true })
   275→    const [result] = await withTimeout(sql`SELECT 1 as ok`, 5_000)
   276→    const ok = result?.ok === 1
   277→    connectionCache.set(slug, { ok, ts: Date.now() })
   278→    return ok
   279→  } catch {
   280→    connectionCache.set(slug, { ok: false, ts: Date.now() })
   281→    return false
   282→  }
   283→}

   300→// ─── Safe query helper ───────────────────────────────────────────────────────
   301→async function safeQuery<T>(
   302→  label: string,
   303→  fn: () => Promise<T[]>,
   304→  fallback: T[] = []
   305→): Promise<T[]> {
   306→  try {
   307→    return await fn()
   308→  } catch (error) {
   309→    console.error(`[ERP-CONNECTOR] ${label} query failed:`, error instanceof Error ? error.message : error)
   310→    return fallback
   311→  }
   312→}
   313→
   314→async function safeQuerySingle<T>(
   315→  label: string,
   316→  fn: () => Promise<T[]>,
   317→  fallback: T
   318→): Promise<T> {
   319→  try {
   320→    const result = await fn()
   321→    return result[0] ?? fallback
   322→  } catch (error) {
   323→    console.error(`[ERP-CONNECTOR] ${label} query failed:`, error instanceof Error ? error.message : error)
   324→    return fallback
   325→  }
   326→}

   328→/** Safely extract a prefix from a potentially null/undefined id value */
   329→function safeIdPrefix(val: any, len: number): string {
   330→  if (val && typeof val === 'string') return val.slice(0, len)
   331→  if (val != null) return String(val).slice(0, len)
   332→  return 'UNKNOWN'
   333→}

   335→// ─── ApparelFlow Data Queries ────────────────────────────────────────────────
   336→async function getApparelFlowStats(): Promise<ErpDashboardStats | null> {
   337→  const sql = await getErpSql('apparelflow')
   338→  if (!sql) return null
   339→
   340→  try {
   341→    // ── Client companies (resilient) ──
   342→    const clients = await safeQuery('AF-Clients', () =>
   343→      sql`
   344→        SELECT
   345→          id, "companyName", "contactPerson", "email", "phone",
   346→          "city", "country", "adminEmail", "licenseType", "licenseStatus",
   347→          "serialKey", "isActive", "databaseStatus", "industry",
   348→          "maxUsers", "maxShops", "selectedModules", "businessType",
   349→          "createdAt"
   350→        FROM "ClientCompany"
   351→        ORDER BY "createdAt" DESC
   352→      `
   353→    )
   354→
   355→    // ── Active clients count ──
   356→    const activeResult = await safeQuerySingle('AF-ActiveClients', () =>
   357→      sql`SELECT COUNT(*)::int AS c FROM "ClientCompany" WHERE "isActive" = true`,
   358→      { c: 0 }
   359→    )
   360→
   361→    // ── Revenue from paid invoices (resilient) ──
   362→    const revenueResult = await safeQuerySingle('AF-Revenue', () =>
   363→      sql`SELECT COALESCE(SUM("totalAmount"), 0)::float AS r FROM "ClientInvoice" WHERE "paymentStatus" = 'paid'`,
   364→      { r: 0 }
   365→    )
   366→
   367→    // ── Total invoices count (resilient) ──
   368→    const invoiceCountResult = await safeQuerySingle('AF-InvoiceCount', () =>
   369→      sql`SELECT COUNT(*)::int AS c FROM "ClientInvoice"`,
   370→      { c: 0 }
   371→    )
   372→
   373→    // ── Client invoices (resilient) ──
   374→    const invoices = await safeQuery('AF-Invoices', () =>
   375→      sql`
   376→        SELECT
   377→          ci.id, ci."invoiceNo", ci."serialKey", ci."licenseType",
   378→          ci."licensePeriod", ci."subtotal", ci."taxAmount", ci."discountAmount",
   379→          ci."totalAmount", ci."paidAmount", ci."paymentStatus",
   380→          ci."startDate", ci."expiryDate", ci."createdAt",
   381→          cc."companyName", cc."adminEmail"
   382→        FROM "ClientInvoice" ci
   383→        LEFT JOIN "ClientCompany" cc ON ci."clientCompanyId" = cc.id
   384→        ORDER BY ci."createdAt" DESC
   385→      `
   386→    )
   387→
   388→    // ── License activations (resilient) ──
   389→    const licenseActivations = await safeQuery('AF-Licenses', () =>
   390→      sql`
   391→        SELECT
   392→          la.id, la."serialKey", la."machineId", la."machineLabel",
   393→          la."ipAddress", la."isActive", la."activatedAt", la."lastCheckedAt",
   394→          cc."companyName", cc."licenseType"
   395→        FROM "LicenseActivation" la
   396→        LEFT JOIN "ClientCompany" cc ON la."clientCompanyId" = cc.id
   397→        ORDER BY la."activatedAt" DESC
   398→      `
   399→    )
   400→
   401→    // ── Revenue by month (last 6 months, resilient) ──
   402→    const revenueByMonth = await safeQuery('AF-RevenueByMonth', () =>
   403→      sql`
   404→        SELECT
   405→          TO_CHAR("createdAt", 'Mon') AS month,
   406→          COALESCE(SUM("totalAmount"), 0)::float AS revenue
   407→        FROM "ClientInvoice"
   408→        WHERE "createdAt" >= NOW() - INTERVAL '6 months'
   409→        GROUP BY TO_CHAR("createdAt", 'Mon'), DATE_TRUNC('month', "createdAt")
   410→        ORDER BY DATE_TRUNC('month', "createdAt") ASC
   411→        LIMIT 6
   412→      `
   413→    )
   414→
   415→    // Map clients to ErpClient format
   416→    const mappedClients: ErpClient[] = clients.map((c: any) => ({
   417→      id: c.id,
   418→      companyName: c.companyName || 'Unknown',
   419→      contactPerson: c.contactPerson || c.adminEmail || '—',
   420→      email: c.email || c.adminEmail || '',
   421→      phone: c.phone || null,
   422→      city: c.city || null,
   423→      country: c.country || 'Pakistan',
   424→      status: c.isActive ? 'active' : 'inactive',
   425→      industry: c.industry || c.businessType || null,
   426→      erpSlug: 'apparelflow',
   427→      erpName: 'ApparelFlow',
   428→      licenseType: c.licenseType || null,
   429→      databaseStatus: c.databaseStatus || null,
   430→      maxUsers: c.maxUsers || null,
   431→      maxShops: c.maxShops || null,
   432→      selectedModules: c.selectedModules || null,
   433→      createdAt: c.createdAt,
   434→    }))
   435→
   436→    // Map invoices
   437→    const mappedInvoices: ErpInvoice[] = invoices.map((inv: any) => ({
   438→      id: inv.id,
   439→      invoiceNumber: inv.invoiceNo || `INV-${safeIdPrefix(inv.id, 8)}`,
   440→      clientName: inv.companyName || 'Unknown',
   441→      clientEmail: inv.adminEmail || null,
   442→      erpSlug: 'apparelflow',
   443→      erpName: 'ApparelFlow',
   444→      amount: inv.subtotal || 0,
   445→      tax: inv.taxAmount || 0,
   446→      totalAmount: inv.totalAmount || 0,
   447→      status: inv.paymentStatus || 'pending',
   448→      dueDate: inv.expiryDate,
   449→      paidAt: inv.paymentStatus === 'paid' ? inv.createdAt : null,
   450→      createdAt: inv.createdAt,
   451→    }))
   452→
   453→    // Map licenses
   454→    const mappedLicenses: ErpLicense[] = licenseActivations.map((lic: any) => ({
   455→      id: lic.id,
   456→      serialKey: lic.serialKey || safeIdPrefix(lic.id, 12).toUpperCase(),
   457→      clientName: lic.companyName || 'Unknown',
   458→      erpSlug: 'apparelflow',
   459→      erpName: 'ApparelFlow',
   460→      licenseType: lic.licenseType || 'standard',
   461→      status: lic.isActive ? 'active' : 'expired',
   462→      deviceName: lic.machineLabel || lic.machineId || null,
   463→      deviceFingerprint: lic.machineId || null,
   464→      activatedAt: lic.activatedAt,
   465→      expiresAt: null,
   466→      createdAt: lic.activatedAt,
   467→    }))
   468→
   469→    return {
   470→      totalClients: clients.length,
   471→      activeClients: activeResult?.c ?? 0,
   472→      totalRevenue: revenueResult?.r ?? 0,
   473→      totalInvoices: invoiceCountResult?.c ?? 0,
   474→      activeLicenses: licenseActivations.length,
   475→      totalDevices: licenseActivations.length,
   476→      clientCompanies: mappedClients,
   477→      invoices: mappedInvoices,
   478→      licenses: mappedLicenses,
   479→      revenueByMonth: revenueByMonth.map((r: any) => ({
   480→        month: r.month,
   481→        revenue: r.revenue,
   482→      })),
   483→    }
   484→  } catch (error) {
   485→    console.error('[ERP-CONNECTOR] ApparelFlow stats failed completely:', error)
   486→    return null
   487→  }
   488→}
