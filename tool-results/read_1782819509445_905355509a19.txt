    48→type CurrencyCode = 'PKR' | 'USD' | 'EUR' | 'GBP' | 'SAR' | 'AED'
    49→
    50→const CURRENCIES: Record<CurrencyCode, { symbol: string; label: string; rate: number }> = {
    51→  PKR: { symbol: 'PKR', label: 'Pakistani Rupee', rate: 1 },
    52→  USD: { symbol: '$', label: 'US Dollar', rate: 0.0036 },
    53→  EUR: { symbol: '€', label: 'Euro', rate: 0.0033 },
    54→  GBP: { symbol: '£', label: 'British Pound', rate: 0.0028 },
    55→  SAR: { symbol: 'SAR', label: 'Saudi Riyal', rate: 0.0135 },
    56→  AED: { symbol: 'AED', label: 'UAE Dirham', rate: 0.0132 },
    57→}
    58→
    59→function formatPrice(pkrPrice: number, currency: CurrencyCode): string {
    60→  const c = CURRENCIES[currency]
    61→  const converted = Math.round(pkrPrice * c.rate)
    62→  if (currency === 'PKR') {
    63→    return converted.toLocaleString('en-PK')
    64→  }
    65→  return converted.toLocaleString('en-US')
    66→}

   446→
   447→function PricingCard({
   448→  tier,
   449→  index,
   450→  isAnnual,
   451→  currency,
   452→  onSelectPlan,
   453→}: {
   454→  tier: (typeof PRICING_TIERS)[number]
   455→  index: number
   456→  isAnnual: boolean
   457→  currency: CurrencyCode
   458→  onSelectPlan: (planId: string) => void
   459→}) {
   460→  const { navigate } = useNavigate()
   461→  const isPopular = tier.popular
   462→  const monthlyPrice = tier.pkrMonthly
   463→  // Annual calculation: 2 months free (pay for 10 months, use for 12)
   464→  // This is more intuitive: "Pay for 10, get 12"
   465→  const annualTotal = monthlyPrice * 10
   466→  const annualMonthlyRate = Math.round(annualTotal / 12)
   467→  const savingsAmount = monthlyPrice * 12 - annualTotal // = monthlyPrice * 2
   468→  const savingsPercent = Math.round((savingsAmount / (monthlyPrice * 12)) * 100) // ~17%
   469→  const price = isAnnual ? annualMonthlyRate : monthlyPrice
   470→  const accent = TIER_ACCENTS[tier.id]
   471→
   472→  const [featuresExpanded, setFeaturesExpanded] = useState(false)
   473→  const INITIAL_FEATURES_COUNT = 4
   474→  const visibleFeatures = featuresExpanded
   475→    ? tier.features
   476→    : tier.features.slice(0, INITIAL_FEATURES_COUNT)
   477→  const hasMoreFeatures = tier.features.length > INITIAL_FEATURES_COUNT
   478→
   479→  return (
   480→    <motion.div
   481→      initial={{ opacity: 0, y: 24 }}
   482→      whileInView={{ opacity: 1, y: 0 }}
   483→      viewport={{ once: true }}
   484→      transition={{ delay: index * 0.15, duration: 0.6 }}
   485→      className={`relative flex flex-col h-full ${isPopular ? 'xl:-mt-4 xl:mb-4 z-10' : ''}`}
   486→    >
   487→      {/* Glow behind popular card — vermillion/amber glow */}
   488→      {isPopular && (
   489→        <div className="absolute -inset-3 rounded-3xl bg-gradient-to-b from-[var(--gg-vermillion)]/20 via-[var(--gg-amber)]/10 to-transparent blur-xl pointer-events-none" />
   490→      )}
   491→
   492→      <InteractiveCard
   493→        glowColor={`var(${tier.accentVar})`}
   494→        variant={isPopular ? 'strong' : 'default'}
   495→        className="flex-1 flex flex-col"
   496→      >
   497→        <div className={`relative bg-card overflow-hidden group flex flex-col h-full ${isPopular ? 'gg-elevated' : 'gg-card'}`}>
   498→          {/* Colored accent bar at top */}
