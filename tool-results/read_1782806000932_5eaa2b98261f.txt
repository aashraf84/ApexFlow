   459→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   460→  const [display, setDisplay] = useState(0)
   461→
   462→  useEffect(() => {
   463→    const duration = 1200
   464→    const startTime = performance.now()
   465→
   466→    function animate(currentTime: number) {
   467→      const elapsed = currentTime - startTime
   468→      const progress = Math.min(elapsed / duration, 1)
   469→      const eased = 1 - Math.pow(1 - progress, 3)
   470→      const val = Math.round(eased * target * 10) / 10
   471→      setDisplay(val)
   472→      if (progress < 1) {
   473→        requestAnimationFrame(animate)
   474→      }
   475→    }
   476→    requestAnimationFrame(animate)
   477→  }, [target])
   478→
   479→  return <span>{display}{suffix}</span>
   480→}
   481→

   483→const springSmooth = { duration: 0.6, ease: [0.25, 0.46, 0.45, 0.94] as const }
